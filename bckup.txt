const express = require('express');
const axios = require('axios');

const app = express();
app.use(express.json({ limit: '50mb' }));

const PORT = 3000;
const TARGET_BASE_URL = 'https://openrouter.ai/api/v1';

// ============================================================
// 1. DAFTAR API KEYS
// ============================================================
const API_KEYS = [
  'sk-or-v1-xxxxx',  // Ganti dengan api keys dari berbagai akun OpenRouter free anda
  'sk-or-v1-yyyyy',
  'sk-or-v1-zzzzz',
];

// ============================================================
// 2. DAFTAR MODEL GRATIS (urutan prioritas fallback)
// ============================================================
const AVAILABLE_MODELS = [
  'minimax/minimax-m3:free',
  'google/gemini-2.0-flash-exp:free',
  'nvidia/nemotron-3-ultra:free',
  'poolside/laguna-s-2.1:free',
  'thinkingmachines/inkling:free',
  'z-ai/glm-5.2:free',
  'cohere/north-mini-code:free',
  'poolside/laguna-xs-2.1:free',
  'qwen/qwen-2.5-72b-instruct:free',
  'microsoft/phi-3.5-mini-128k-instruct:free',
  'meta-llama/llama-3.2-3b-instruct:free',
  'nvidia/nemotron-3.5-lightning:free',
  'inclusionai/ling-3.0-flash-fin:free',
  'dots-studio/dots3-note-preview:free',
];

// ============================================================
// 3. STATE ROTASI
// ============================================================
let currentKeyIndex = 0;
let currentModelIndex = 0;

// ============================================================
// 4. MIDDLEWARE PROXY DENGAN DUKUNGAN STREAMING
// ============================================================
app.use('/v1', async (req, res) => {
  const maxAttempts = API_KEYS.length * AVAILABLE_MODELS.length * 2;
  let attempts = 0;
  
  let tempKeyIndex = currentKeyIndex;
  let tempModelIndex = currentModelIndex;

  while (attempts < maxAttempts) {
    const key = API_KEYS[tempKeyIndex];
    const model = AVAILABLE_MODELS[tempModelIndex];

    console.log(`🔄 Coba: Key[${tempKeyIndex+1}/${API_KEYS.length}] + Model: ${model}`);

    try {
      // === Siapkan payload, timpa model ===
      const payload = {
        ...req.body,
        model: model,
      };

      // Deteksi apakah request meminta streaming
      const isStreaming = req.body.stream === true;

      // === Konfigurasi axios ===
      const axiosConfig = {
        method: req.method,
        url: TARGET_BASE_URL + req.url,
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${key}`,
          'HTTP-Referer': 'http://localhost:3000',
          'X-Title': 'Claude Code Proxy (Auto-Rolling)',
        },
        data: payload,
        timeout: 60000,
        responseType: isStreaming ? 'stream' : 'json',
      };

      const response = await axios(axiosConfig);

      // === SUKSES ===
      currentKeyIndex = tempKeyIndex;
      currentModelIndex = tempModelIndex;
      console.log(`✅ Sukses! Key[${tempKeyIndex+1}] Model: ${model}`);

      // === Jika streaming, pipe langsung ke client ===
      if (isStreaming) {
        // Set header dari response (termasuk content-type, dll)
        res.set(response.headers);
        // Pipe stream dari OpenRouter ke client
        response.data.pipe(res);
        return;
      }

      // === Non-streaming: kirim JSON biasa ===
      return res.status(response.status).json(response.data);

    } catch (error) {
      const status = error.response?.status || 500;
      const errorMsg = error.response?.data?.error?.message || error.message;

      // === GAGAL KARENA LIMIT / KEY INVALID ===
      if (status === 429 || status === 401 || status === 403 || errorMsg.includes('rate limit')) {
        console.warn(`⚠️ Gagal (${status}) Key[${tempKeyIndex+1}]. Pindah key...`);
        tempKeyIndex = (tempKeyIndex + 1) % API_KEYS.length;
        attempts++;

        if (tempKeyIndex === 0) {
          console.log(`🔄 Semua key gagal untuk model ${model}. Pindah ke model berikutnya...`);
          tempModelIndex = (tempModelIndex + 1) % AVAILABLE_MODELS.length;
          tempKeyIndex = 0;
        }
        continue;
      }

      // === ERROR MODEL ===
      if (status >= 500 || errorMsg.includes('model') || errorMsg.includes('overloaded') || errorMsg.includes('not found')) {
        console.warn(`⚠️ Error ${status} pada model ${model}. Pindah ke model berikutnya...`);
        tempModelIndex = (tempModelIndex + 1) % AVAILABLE_MODELS.length;
        tempKeyIndex = 0;
        attempts++;
        continue;
      }

      // === ERROR TAK TERDUGA ===
      console.error(`💥 Error fatal:`, errorMsg);
      return res.status(status).json(error.response?.data || { error: errorMsg });
    }
  }

  // === SEMUA HABIS ===
  res.status(429).json({
    error: '❌ Semua kombinasi Key dan Model gagal. Coba lagi nanti.',
  });
});

// ============================================================
// 5. JALANKAN SERVER
// ============================================================
app.listen(PORT, () => {
  console.log(`✅ Proxy Rolling (dengan streaming) berjalan di http://localhost:${PORT}`);
  console.log(`📌 ${API_KEYS.length} Keys siap, ${AVAILABLE_MODELS.length} Model siap.`);
  console.log(`📌 Tekan CTRL+C untuk menghentikan proxy.`);
});
